REMOTE_DATA = "docs/ganesha_data.json"


def _build_snap(state):
    closed = state.get("closed", [])
    wins = [t for t in closed if t.get("pnl_net", 0) > 0]
    losses = [t for t in closed if t.get("pnl_net", 0) <= 0]
    n = len(closed)
    win_rate = (len(wins) / n * 100) if n else 0.0
    gross_win = sum(t["pnl_net"] for t in wins)
    gross_loss = abs(sum(t["pnl_net"] for t in losses))
    profit_factor = (gross_win / gross_loss) if gross_loss > 0 else (gross_win and 999 or 0)

    eq = []
    running = 0.0
    for t in closed:
        running += t.get("pnl_net", 0)
        eq.append({"ts": t.get("closed_ts"), "cum_pnl": round(running, 2)})

    return {
        "updated_at": int(time.time()),
        "bot": "Ganesha",
        "dry_run": C.DRY_RUN,
        "equity_now": state.get("equity_now"),
        "realized_pnl": round(state.get("realized_pnl", 0), 2),
        "deposits_total": round(state.get("deposits_total", 0), 2),
        "main_spot_usdt": state.get("main_spot_usdt"),
        "trades_total": n,
        "win_rate": round(win_rate, 1),
        "profit_factor": round(profit_factor, 2),
        "trades_today": state.get("daily", {}).get("trades_today", 0),
        "pnl_today": round(state.get("daily", {}).get("pnl_today", 0), 2),
        "open_positions": [
            {"symbol": s, "entry": p.get("entry"), "qty": p.get("qty"),
             "stop": p.get("stop"), "tp1_done": p.get("tp1_done"),
             "opened_ts": p.get("opened_ts")}
            for s, p in state.get("positions", {}).items()
            if p.get("status") == "open"
        ],
        "recent_closed": closed[-15:],
        "equity_curve": eq,
    }


def write_snapshot(state):
    """Escribe snapshot local y lo pushea al repo. Nunca lanza."""
    snap = _build_snap(state)
    try:
        with open(C.SNAPSHOT, "w") as f:
            json.dump(snap, f, indent=2, default=str)
    except Exception as e:
        print(f"[dashboard] write local fallo: {e}")
    try:
        payload = json.dumps(snap, ensure_ascii=False, indent=1, default=str).encode("utf-8")
        _gh_publish(REMOTE_DATA, payload, f"ganesha: snapshot {snap['updated_at']}")
        print("[dashboard] data publicada OK")
    except Exception as e:
        print(f"[dashboard] push fallo (ignorado): {e}")
    return snap


# ---------- GitHub Contents API (stdlib) ----------
def _gh_headers():
    return {
        "Authorization": f"Bearer {GH_TOKEN}",
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
        "User-Agent": "ganesha-bot",
    }


def _gh_get_sha(path):
    url = f"{GH_API}/repos/{GH_REPO}/contents/{path}?ref={GH_BRANCH}"
    req = urllib.request.Request(url, headers=_gh_headers())
    try:
        with urllib.request.urlopen(req, timeout=20) as r:
            return json.load(r).get("sha")
    except urllib.error.HTTPError as e:
        if e.code == 404:
            return None
        raise


def _gh_publish(path, content_bytes, message):
    if not GH_TOKEN:
        raise RuntimeError("GITHUB_TOKEN no seteado en .env")
    sha = _gh_get_sha(path)
    body = {
        "message": message,
        "content": base64.b64encode(content_bytes).decode(),
        "branch": GH_BRANCH,
    }
    if sha:
        body["sha"] = sha
    url = f"{GH_API}/repos/{GH_REPO}/contents/{path}"
    req = urllib.request.Request(
        url, data=json.dumps(body).encode(),
        headers={**_gh_headers(), "Content-Type": "application/json"},
        method="PUT",
    )
    with urllib.request.urlopen(req, timeout=20) as r:
        return r.status


def gh_put_path(remote_path, local_file, message):
    """Helper para deploy puntual de archivos (html, favicon, etc.)."""
    with open(local_file, "rb") as f:
        data = f.read()
    return _gh_publish(remote_path, data, message)

from freqtrade.strategy import IStrategy
from pandas import DataFrame


class DonchianDaily(IStrategy):
    INTERFACE_VERSION = 3
    timeframe = '1d'
    can_short = False
    # Salida primaria = senal Donchian (minimo 10 dias). El stop -30% es red dura de
    # catastrofe, colocada nativa en Binance. Backtest confirma que nunca dispara antes
    # que la salida Donchian (OOS identico con y sin el: +44.5%).
    stoploss = -0.30
    minimal_roi = {"0": 100}
    process_only_new_candles = True
    use_exit_signal = True
    exit_profit_only = False
    trailing_stop = False
    startup_candle_count = 35

    order_types = {
        'entry': 'limit',
        'exit': 'limit',
        'stoploss': 'limit',
        'stoploss_on_exchange': True,
        'stoploss_on_exchange_interval': 60,
        'stoploss_on_exchange_limit_ratio': 0.99,
    }

    entry_n = 30
    exit_m = 10

    def populate_indicators(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe['don_hi'] = dataframe['close'].rolling(self.entry_n).max().shift(1)
        dataframe['don_lo'] = dataframe['close'].rolling(self.exit_m).min().shift(1)
        return dataframe

    def populate_entry_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe.loc[dataframe['close'] > dataframe['don_hi'], 'enter_long'] = 1
        return dataframe

    def populate_exit_trend(self, dataframe: DataFrame, metadata: dict) -> DataFrame:
        dataframe.loc[dataframe['close'] < dataframe['don_lo'], 'exit_long'] = 1
        return dataframe
